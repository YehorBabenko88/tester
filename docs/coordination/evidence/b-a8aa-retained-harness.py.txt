    def setUp(self):
        super().setUp()

    def test_master_original_future_reproduction(self):
        start,end=1700000000000,4102444800000
        run='master-future:'+self.symbol
        record=self.record();record['source']['ingestion_run']=run
        evidence=self.ev(start_ms=start,end_ms=end,ingestion_run=run,
            pages=[dict(params={'start':start,'end':end-1},cursor=None,next_cursor=None,count=1)],
            available_range=dict(start_ms=start,end_ms=end,method='operator_attestation',
                                 source_url='https://example.test/future-manifest',sha256='a'*64))
        self.store.ingest([record],evidence,self.symbol,run_id=run)
        page=self.store.read_page('trades','bybit','linear',self.symbol,start,end)
        self.assertEqual(len(page['rows']),1)
        self.assertEqual(page['coverage']['end_ms'],end)
        self.assertFalse(page['coverage']['complete'])
        self.assertEqual(page['coverage']['proof_refs'],[])
        self.assertLess(page['snapshot']['clock_ms'],end)
        report['master_original_repro']={'coverage':page['coverage'],'snapshot':page['snapshot']}

    def test_live_knowledge_and_c_rejection(self):
        self.ingest('knowledge')
        page=self.page()
        self.assertTrue(page['coverage']['complete'])  # current-time positive control
        knowledge=page['knowledge_time']
        self.assertEqual(knowledge['status'],'UNAVAILABLE')
        self.assertIsNone(knowledge['quality_available_at_ms'])
        self.assertFalse(knowledge['historical_as_of'])
        self.assertTrue(knowledge['facts'])
        observation=page['observations'][0]
        with self.store.connect() as connection:
            original=connection.execute('SELECT to_jsonb(recorded_at) AS stamp FROM tester_storage.source_observations WHERE observation_id=%s',(observation['observation_id'],)).fetchone()['stamp']
        self.assertEqual(observation['observation_recorded_at'],original)
        facts=[f for f in knowledge['facts'] if f['identity'].get('observation_id')==observation['observation_id']]
        self.assertTrue(facts)
        self.assertEqual(facts[0]['timestamps']['observation_recorded_at'],original)
        from src.analytics.binding import BindingUnavailable,bind_storage_trades
        with self.assertRaises(BindingUnavailable):
            bind_storage_trades(page['rows'],venue='bybit',category='linear',symbol=self.symbol,
                start_ms=1700000000000,end_ms=1700000060000,
                information_cutoff_ms=page['snapshot']['clock_ms'],quality_available_at_ms=None,
                availability={'status':'AVAILABLE','reason':None},
                coverage={'status':'UNKNOWN','start_ms':1700000000000,'end_ms':1700000060000,'evidence_refs':[]})
        self.ingest('late-knowledge')
        capped=self.page(max_observations=1)
        self.assertEqual(capped['outcome'],'RESOURCE_EXHAUSTED')
        self.assertEqual(capped['coverage']['state'],'UNKNOWN')
        self.assertEqual(capped['coverage']['proof_refs'],[])
        self.assertEqual(capped['knowledge_time']['facts'],[])
        report['live_knowledge']={'current_coverage_positive_control':True,'historical_status':'UNAVAILABLE',
