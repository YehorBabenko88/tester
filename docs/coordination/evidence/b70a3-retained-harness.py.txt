def pg_command(pg, arguments, allowed=(0,)):
    """Resident PG children must never inherit a PIPE handle."""
    env = {k: v for k, v in os.environ.items() if not k.upper().startswith("PG")}
    result = subprocess.run(
        [str(pg.PGBIN / "pg_ctl.exe"), *map(str, arguments)],
        stdin=subprocess.DEVNULL,
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,
        timeout=45,
        env=env,
        creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0),
    )
    if result.returncode not in allowed:
        raise RuntimeError("owned_pg_command_failed")
    return result.returncode


class SafeResult(unittest.TestResult):
    """Persist only case names/classes/sqlstate; never format driver exceptions/DSNs."""

    def __init__(self):
        super().__init__()
        self.cases = []

    def addSuccess(self, test):
        super().addSuccess(test)
        self.cases.append(dict(case=test._testMethodName, status="PASS"))

    def addError(self, test, error):
        super().addError(test, error)
        self.record(test, error, "ERROR")

    def addFailure(self, test, error):
        super().addFailure(test, error)
        self.record(test, error, "FAIL")

    def record(self, test, error, status):
        self.cases.append(
            dict(
                case=test._testMethodName,
                status=status,
                error_type=error[0].__name__,
                sqlstate=getattr(error[1], "sqlstate", None),
            )
        )

    def test_07_actual_transaction_exit_keeps_gate_until_commit(self):
        f = self.context["foundation"]
        epoch = self.epoch("commit")
        other = self.epoch("commit-other")
        reached = threading.Event()
        release = threading.Event()
        factory = self.factory

        class PausedCommit:
            def __init__(self):
                self.raw = factory()

            def __getattr__(self, name):
                return getattr(self.raw, name)

            @contextmanager
            def transaction(self):
                with self.raw.transaction():
                    yield
                    reached.set()
                    if not release.wait(3):
                        raise RuntimeError("synthetic_commit_barrier_timeout")

        def writer():
            with f.experimental_mutation(
                PausedCommit, epoch, ["commit"], ["raw_records"]
            ) as c:
                c.execute(
                    "INSERT INTO tester_foundation_probe.facts(id) VALUES(%s)",
                    ("fixture:commit-body",),
                )

        with ThreadPoolExecutor(max_workers=1) as pool:
            future = pool.submit(writer)
            try:
                self.assertTrue(reached.wait(3))
                self.assertEqual(self.fixture_count("fixture:commit-body"), 0)
                with self.assertRaisesRegex(RuntimeError, "gate busy"):
                    with f.experimental_mutation(
                        self.factory, other, ["other"], ["raw_records"]
                    ):
                        self.fail("gate released before commit")
            finally:
                release.set()
            future.result(timeout=3)
        self.assertEqual(self.fixture_count("fixture:commit-body"), 1)
        with f.experimental_mutation(self.factory, other, ["other"], ["raw_records"]):
            pass
