def sample(label):
    parent=psutil.Process(int((pg.DATA/'postmaster.pid').read_text().splitlines()[0]))
    processes=[parent,*parent.children(recursive=True)]
    return {'label':label,'pg_processes':len(processes),
            'pg_rss_sum_bytes_includes_shared_pages':sum(p.memory_info().rss for p in processes),
            'pg_cpu_seconds_sum':sum(sum(p.cpu_times()[:2]) for p in processes),
